cmake_minimum_required(VERSION 3.20)
project(open_elbow_firmware C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_library(oee_control src/control.c src/protocol.c)
target_include_directories(oee_control PUBLIC include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(oee_control PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
endif()
add_executable(host_sim src/host_sim.c)
target_link_libraries(host_sim PRIVATE oee_control)
add_executable(control_tests tests/control_tests.c)
target_link_libraries(control_tests PRIVATE oee_control)
if(NOT WIN32)
  target_link_libraries(host_sim PRIVATE m)
  target_link_libraries(control_tests PRIVATE m)
endif()
add_test(NAME control_fault_protocol_regressions COMMAND control_tests)
